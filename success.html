Same render, via curl

SPEC='{"type":"bar","title":"It works!","data":{"labels":["Mon","Tue","Wed"],
       "series":[{"values":[12,19,8]}]}}'

curl "https://api.slickfast.com/chart.png?spec=$(jq -rn --arg s "$SPEC" '$s|@uri')" \
  -H "Authorization: Bearer sf_your_key_here" \
  --output first-chart.png && open first-chart.png
